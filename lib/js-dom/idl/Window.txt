[Kind=window, Global]
interface Window : EventTarget {
  [Replaceable] readonly attribute any window;
  [Replaceable] readonly attribute any self;
  [Replaceable] readonly attribute any frames;
  [Replaceable] readonly attribute any globalThis;
  readonly attribute any document;
  attribute DOMString name;
  [PutForwards=href] readonly attribute any location;
  readonly attribute any history;
  readonly attribute any navigator;
  [Replaceable] readonly attribute any screen;
  [Replaceable] readonly attribute any performance;
  [Replaceable] readonly attribute any parent;
  [Replaceable] readonly attribute any top;
  readonly attribute any opener;
  readonly attribute any frameElement;
  [Replaceable] readonly attribute unsigned long length;
  readonly attribute boolean closed;
  [Replaceable] readonly attribute any event;
  readonly attribute any localStorage;
  readonly attribute any sessionStorage;
  readonly attribute any customElements;
  readonly attribute boolean isSecureContext;
  readonly attribute USVString origin;

  undefined close();
  undefined stop();
  undefined focus();
  undefined blur();
  any open(optional USVString url = "", optional DOMString target = "_blank", optional DOMString features = "");

  undefined alert(optional any message);
  boolean confirm(optional DOMString message = "");
  DOMString? prompt(optional DOMString message = "", optional DOMString default = "");
  undefined print();
  undefined postMessage(any message, optional any targetOrigin, optional any transfer);

  // CSSOM View.
  [Throws] any matchMedia(DOMString query);
  [Replaceable] readonly attribute long innerWidth;
  [Replaceable] readonly attribute long innerHeight;
  [Replaceable] readonly attribute long outerWidth;
  [Replaceable] readonly attribute long outerHeight;
  [Replaceable] readonly attribute double scrollX;
  [Replaceable, Impl=Window_scrollX] readonly attribute double pageXOffset;
  [Replaceable] readonly attribute double scrollY;
  [Replaceable, Impl=Window_scrollY] readonly attribute double pageYOffset;
  [Replaceable] readonly attribute long screenX;
  [Replaceable, Impl=Window_screenX] readonly attribute long screenLeft;
  [Replaceable] readonly attribute long screenY;
  [Replaceable, Impl=Window_screenY] readonly attribute long screenTop;
  [Replaceable] readonly attribute double devicePixelRatio;
  undefined scroll(optional any x, optional any y);
  [Impl=Window_scroll] undefined scrollTo(optional any x, optional any y);
  undefined scrollBy(optional any x, optional any y);
  undefined moveTo(long x, long y);
  undefined resizeTo(long width, long height);

  [Throws] any getComputedStyle(Element elt, optional DOMString? pseudoElt);
  any getSelection();

  [Throws] unsigned long requestAnimationFrame(FrameRequestCallback callback);
  undefined cancelAnimationFrame(unsigned long handle);
  [Throws] unsigned long requestIdleCallback(VoidFunction callback, optional any options);
  undefined cancelIdleCallback(unsigned long handle);

  [Throws] DOMString atob(DOMString data);
  [Throws] DOMString btoa(DOMString data);
  undefined reportError(any e);
  [Throws] any structuredClone(any value, optional any options);
};
Window includes GlobalEventHandlers;
Window includes WindowEventHandlers;
