// CSSOM: CSSStyleDeclaration (element.style over the style attribute, and
// getComputedStyle's read-only one) and the CSS namespace. The camelCase
// and dashed property accessors (style.backgroundColor) are defined by the
// implementation from lib/css's property list (jsdom_style.swift).

[Kind=object, Swift=JSDOMStyleDeclaration, Exotic]
interface CSSStyleDeclaration {
  [CEReactions, Throws] attribute DOMString cssText;
  readonly attribute unsigned long length;
  getter DOMString item(unsigned long index);
  DOMString getPropertyValue(DOMString property);
  DOMString getPropertyPriority(DOMString property);
  [CEReactions, Throws] undefined setProperty(DOMString property, [LegacyNullToEmptyString] DOMString value, [LegacyNullToEmptyString] optional DOMString priority = "");
  [CEReactions, Throws] DOMString removeProperty(DOMString property);
  readonly attribute any parentRule;
  [CEReactions, LegacyNullToEmptyString, Throws] attribute DOMString cssFloat;
};

// CSSOM: constructable stylesheets (new CSSStyleSheet(), replace /
// replaceSync, insertRule / deleteRule, cssRules), adopted by documents
// and shadow roots (DocumentOrShadowRoot.adoptedStyleSheets) and handed
// to lib/layout. Rules are kept as text and parsed by lib/css
// (jsdom_stylesheets.swift).
[Kind=object, Swift=JSDOMStyleSheet]
interface StyleSheet {
};

[Kind=object, Swift=JSDOMStyleSheet]
interface CSSStyleSheet : StyleSheet {
  [Throws] constructor(optional any options);
  readonly attribute DOMString type;
  readonly attribute DOMString? href;
  readonly attribute any ownerNode;
  readonly attribute any parentStyleSheet;
  readonly attribute DOMString? title;
  readonly attribute any media;
  attribute boolean disabled;
  readonly attribute any ownerRule;
  [SameObject] readonly attribute CSSRuleList cssRules;
  [SameObject, Impl=CSSStyleSheet_cssRules] readonly attribute CSSRuleList rules;
  [Throws] unsigned long insertRule(DOMString rule, optional unsigned long index = 0);
  [Throws] undefined deleteRule(unsigned long index);
  [Throws] long addRule(optional DOMString selector = "undefined", optional DOMString style = "undefined", optional any index);
  [Throws] undefined removeRule(optional unsigned long index = 0);
  any replace(DOMString text);
  [Throws] undefined replaceSync(DOMString text);
};

[Kind=object, Swift=JSDOMCSSRuleList, Exotic]
interface CSSRuleList {
  getter CSSRule? item(unsigned long index);
  readonly attribute unsigned long length;
};

// One interface for every rule kind (type tells them apart).
[Kind=object, Swift=JSDOMCSSRule]
interface CSSRule {
  const unsigned short STYLE_RULE = 1;
  const unsigned short CHARSET_RULE = 2;
  const unsigned short IMPORT_RULE = 3;
  const unsigned short MEDIA_RULE = 4;
  const unsigned short FONT_FACE_RULE = 5;
  const unsigned short PAGE_RULE = 6;
  const unsigned short KEYFRAMES_RULE = 7;
  const unsigned short KEYFRAME_RULE = 8;
  const unsigned short NAMESPACE_RULE = 10;
  const unsigned short SUPPORTS_RULE = 12;
  attribute DOMString cssText;
  readonly attribute unsigned short type;
  readonly attribute any parentRule;
  readonly attribute any parentStyleSheet;
  readonly attribute DOMString selectorText;
};

namespace CSS {
  boolean supports(DOMString property, optional DOMString value);
  DOMString escape(DOMString ident);
};
